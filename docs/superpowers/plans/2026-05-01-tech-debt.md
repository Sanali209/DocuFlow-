### Task 2: Batch-query + thread offload в NSMirrorService

**Проблема:** `_sync_bucket` (1) делает N+1 запросов `session.get(TaskItem, id)` в цикле; (2) синхронный `with Session(...)` блок внутри async; (3) `_calculate_md5` читает файл в event loop.

```python
    def _db_load_bucket_tasks(self, node_id: str) -> list[TaskItem]:
        """Sync: load all TaskItems for this node's bucket entries in one JOIN query."""
        with Session(self.db_engine) as session:
            return list(
                session.exec(
                    select(TaskItem)
                    .join(WorkerBucketEntry, TaskItem.id == WorkerBucketEntry.task_item_id)
                    .where(WorkerBucketEntry.node_id == node_id)
                ).all()
            )

    def _db_commit_logs(self, logs: list[WorkLog]) -> None:
        """Sync: persist a batch of WorkLog entries."""
        if not logs:
            return
        with Session(self.db_engine) as session:
            for log in logs:
                session.add(log)
            session.commit()
```

```python
    async def _sync_bucket(self, settings: FolderScannerSettings) -> None:
        """Fetch tasks in bucket and mirror them."""
        # Load tasks via thread to avoid blocking the event loop
        active_tasks: list[TaskItem] = await anyio.to_thread.run_sync(
            self._db_load_bucket_tasks, self.config.node_id
```

```python
        self, task: TaskItem, settings: FolderScannerSettings
    ) -> list[WorkLog]:
        """Ensure a single task is correctly mirrored. Returns any WorkLog entries to persist."""
        logs: list[WorkLog] = []

        src_path: Path | None = self._resolve_source_path_no_session(task, settings)
        if not src_path or not src_path.exists():
            logger.error(f"Source file not found for task {task.file_name}: {src_path}")
            return logs

        dst_path: Path = Path(settings.local_ns_path) / task.file_path

        if not dst_path.exists():
            await self._copy_file(src_path, dst_path, settings.ns_mirror_copy_timeout_s)
            logs.append(self._make_log(task, f"Copied to NS: {task.file_name}"))
            return logs

        # MD5 check via thread (avoids blocking event loop on local file read)
        local_md5: str = await anyio.to_thread.run_sync(self._calculate_md5, dst_path)
        if task.file_hash and local_md5 != task.file_hash:
            logger.warning(f"MD5 Mismatch for {task.file_name}: Network MD5 has changed.")
            logs.append(
                self._make_log(
                    task,
                    "⚠️ Сетевой файл обновился. Локальная копия устарела!",
                    log_type=WorkLogType.FILE_CHANGED,
                )
            )

        return logs
```

```python
        """Merge a remote snapshot into the local database.

        Args:

        if not snapshot_path.exists():

        try:
```

```python
"""Shared file I/O utilities for DocuFlow infrastructure.

Note: MD5 is used here only for fast file comparison and deduplication,
not for cryptographic security purposes.
"""

import hashlib
from pathlib import Path
from typing import BinaryIO

# Chunk size for streaming file reads (256 KiB balances throughput and memory)
FILE_MD5_CHUNK_SIZE: int = 262144


def calculate_file_md5(path: Path) -> str:
    """Calculate MD5 hex-digest of a file using streaming reads.

    Args:
        path: Path to the file. Must exist and be readable.

    Returns:
        Lowercase hex string of the MD5 digest.
    """
    h: hashlib._Hash = hashlib.md5()  # noqa: S324
    f: BinaryIO
    with open(path, "rb") as f:
        chunk: bytes
        for chunk in iter(lambda: f.read(FILE_MD5_CHUNK_SIZE), b""):
            h.update(chunk)
    return h.hexdigest()
```
